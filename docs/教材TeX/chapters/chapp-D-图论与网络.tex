\section{D.1
直觉：图表示什么}\label{d.1-ux76f4ux89c9ux56feux8868ux793aux4ec0ux4e48}

\begin{itemize}
\tightlist
\item
  \textbf{节点（顶点）}：实体（人、城市、网页、商品）。
\item
  \textbf{边}：关系（朋友、道路、超链接、共同购买）。有向边（关注）与无向边（好友）、加权边（距离/强度）。
\item
  图模型能表达''结构''：社团、传播、最短路、枢纽------这正是 NetworkX
  章要做的事。
\end{itemize}

\section{D.3
经典算法：思路与用途}\label{d.3-ux7ecfux5178ux7b97ux6cd5ux601dux8defux4e0eux7528ux9014}

\subsection{D.3.1 遍历}\label{d.3.1-ux904dux5386}

\begin{itemize}
\tightlist
\item
  \textbf{BFS（广度）}：按''层''扩散，找最短路（无权图）。
\item
  \textbf{DFS（深度）}：一条路走到黑，检测环、连通分量、拓扑排序。
\end{itemize}

\subsection{D.3.3
树与网络流}\label{d.3.3-ux6811ux4e0eux7f51ux7edcux6d41}

\begin{itemize}
\tightlist
\item
  \textbf{最小生成树（MST）}：连接所有点且总权重最小（Kruskal 按边排序 +
  并查集；Prim 加点）。
\item
  \textbf{最大流/最小割}：二分匹配、交通流（Ford--Fulkerson）。
\end{itemize}

\subsection{D.3.4
中心性与社区}\label{d.3.4-ux4e2dux5fc3ux6027ux4e0eux793eux533a}

\begin{itemize}
\tightlist
\item
  \textbf{度中心性}：连接的边数。
\item
  \textbf{接近中心性}：到所有点的平均距离（信息传播快慢）。
\item
  \textbf{介数中心性}：经过该点的最短路径数量（桥梁/瓶颈）。
\item
  \textbf{PageRank}：有向图随机游走稳态（重要性传播），是搜索引擎核心向量。
\item
  \textbf{社区发现}：把网络分成''内部紧密、外部稀疏''的组（Louvain、谱聚类、标签传播）。
\end{itemize}

\subsection{微积分}\label{ux5faeux79efux5206}

\begin{itemize}
\tightlist
\item
  \texttt{(x\^{}n)\textquotesingle{}\ =\ n\ x\^{}\{n-1\}}；\texttt{(e\^{}x)\textquotesingle{}\ =\ e\^{}x}；\texttt{(ln\ x)\textquotesingle{}\ =\ 1/x}
\item
  泰勒：\texttt{f(x+h)\ ≈\ f(x)\ +\ f\textquotesingle{}(x)h\ +\ f\textquotesingle{}\textquotesingle{}(x)h²/2}
\item
  数值微分（中心差分）：\texttt{f\textquotesingle{}(x)\ ≈\ (f(x+h)\ -\ f(x-h))\ /\ (2h)}
\end{itemize}

\subsection{线性代数}\label{ux7ebfux6027ux4ee3ux6570}

\begin{itemize}
\tightlist
\item
  \texttt{(AB)\^{}T\ =\ B\^{}T\ A\^{}T}；\texttt{tr(AB)\ =\ tr(BA)}
\item
  最小二乘：\texttt{x\^{}\ =\ (A\^{}T\ A)\^{}\{-1\}\ A\^{}T\ b}（数值上多用
  \texttt{lstsq}）
\item
  SVD：\texttt{A\ =\ UΣV\^{}T}；对称矩阵：\texttt{A\ =\ QΛQ\^{}T}
\end{itemize}

\subsection{概率统计}\label{ux6982ux7387ux7edfux8ba1}

\begin{itemize}
\tightlist
\item
  \texttt{E{[}X{]}\ =\ Σ\ x\ p(x)}；\texttt{Var(X)\ =\ E{[}X²{]}\ -\ (E{[}X{]})²}
\item
  样本方差：\texttt{s²\ =\ Σ(x\_i\ -\ x̄)²\ /\ (n-1)}
\item
  置信区间：\texttt{x̄\ ±\ t\_\{α/2,\ n-1\}\ ·\ s/√n}
\item
  线性回归：\texttt{y\ =\ Xβ\ +\ ε}，\texttt{β\^{}\ =\ (X\^{}T\ X)\^{}\{-1\}\ X\^{}T\ y}，\texttt{R²\ =\ 1\ -\ SSE/SST}
\end{itemize}

\subsection{机器学习}\label{ux673aux5668ux5b66ux4e60}

\begin{itemize}
\tightlist

  逻辑回归（sigmoid）：\texttt{p\ =\ 1\ /\ (1\ +\ e\^{}\{-θ\^{}T\ x\})}；损失

\section{F.4
数值稳定性（科学计算必修，讲解）}\label{f.4-ux6570ux503cux7a33ux5b9aux6027ux79d1ux5b66ux8ba1ux7b97ux5fc5ux4feeux8bb2ux89e3}

浮点数是有限精度的（约 15\textasciitilde16
位有效小数），所以''数学上对''的公式在电脑上可能''崩''：

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{比较}：别用 \texttt{==} 比浮点；用 \texttt{np.isclose} /
  \texttt{abs(a-b)\ \textless{}\ 1e-9}。
\item
  \textbf{条件数}：\texttt{np.linalg.cond(A)} 大 →
  病态，解对输入过敏；优先看它再''硬解''。
\item
  \textbf{灾难性消去}：\texttt{sqrt(x²+1)\ -\ x} 在大 \texttt{x}
  时相减丢精度，改写 \texttt{1/(sqrt(x²+1)+x)}。
\item
  \textbf{缩放}：优化/距离/聚类前对特征标准化（\texttt{StandardScaler}）。
\item
  \textbf{对数域}：连乘小概率时取 log，避免下溢；分类看概率用
  \texttt{log\_proba}。
\item
  \textbf{可复现}：固定种子（\texttt{random\_state} /
  \texttt{np.random.default\_rng(seed)}）。
\end{enumerate}

配一个手算直觉：\texttt{x\ =\ 10\^{}8} 时，\texttt{sqrt(x²+1)\ -\ x}
用双精度算出来≈\texttt{0}（真正约
\texttt{5×10\^{}\{-9\}}）------不是数学错了，是浮点''看不见''那么小的差。
